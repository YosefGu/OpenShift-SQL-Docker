
-- create a persistent volume -- 
oc apply -f - <<EOF 
  apiVersion: v1
  kind: PersistentVolumeClaim
  metadata:
    name: mysql-pvc
  spec:
    accessModes:
      - ReadWriteMany
    resources:
      requests:
        storage: 1Gi
EOF

-- create a sql server -- 
oc new-app quay.io/sclorg/mysql-80-c9s \
    --name=mysql-server \
    -e MYSQL_ROOT_PASSWORD=myrootpass \
    -e MYSQL_USER=username \
    -e MYSQL_PASSWORD=password \
    -e MYSQL_DATABASE=database \
    --mount pvc=mysql-pvc,container-path=/var/lib/mysql


-- expose the server --
oc expose service/mysql<id>

-- entering to pod --
oc get pods
oc rsh <pod id>

-- entering the mysql server --
mysql --user=username --password database

-- createing a table into database --
CREATE TABLE users (id int, first_name varchar(255), last_name varchar(255)) ;

-- adding data to table --
INSERT INTO users (id, first_name, last_name) VALUES (1, 'Avi', 'Cohen') ;

-- creating un image from my repo --
oc new-app https://github.com/YosefGu/OpenShift-SQL-Docker \
    --name=python-app \
    -e MYSQL_HOST=mysql-80-c9s \
    -e MYSQL_USER=username \
    -e MYSQL_PASSWORD=password \
    -e MYSQL_DATABASE=database

-- create a service for python server container --
oc expose deployment/python-app --port=8000 --target-port=8000 --name=python-app-svc

-- expose the service --
oc expose svc/python-app-svc
